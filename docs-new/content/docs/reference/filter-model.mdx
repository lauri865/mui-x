Example of a filter model with multiple conditions and logic operators – matches a `person` with the name `John` or `Jane` and `age` greater than or equal to 18:
```ts twoslash
  import { GridFilterModel, GridLogicOperator } from '@mui/x-data-grid';
  // ---cut---
  const filterModel: GridFilterModel = {
    logicOperator: 'and',
    items: [
      {
        field: 'name',
        logicOperator: 'and',
        conditions: [
          { 
            operator: 'equals',
            value: 'John',
          },
          {
            operator: 'equals',
            value: 'Jane',
          },
        ],
      },
      {
        field: 'age',
        conditions: [
          {
            operator: '>=',
            value: 18,
          },
        ],
      },
    ],
  };
```

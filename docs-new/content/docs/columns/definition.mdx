```tsx twoslash
import React from 'react';
const rows = [{id: 1, name: 'John Doe', joined_at: new Date()}];
type RowModel = (typeof rows)[number];
// ---cut---
import { DataGrid, createColumns } from '@mui/x-data-grid';

const props = createColumns<RowModel>((col)=>[
  col.string({ field: "name", headerName: "Name", width: 150, pinned: "left" }),
// @noErrors
  col.d
//     ^|
]);





<DataGrid rows={[]} {...props} />>
```
